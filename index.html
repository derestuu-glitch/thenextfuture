Agar email peserta, upgrade paket, dan hasil tes CAT bisa tersimpan online dari semua perangkat tanpa terblokir, buka Firebase Console → Firestore Database → Rules lalu tempel aturan berikut dan klik Publish: 

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /authorized_users/{email} {
      allow read, write: if true;

    match /tnf_cat_results/{resultId} {
      allow read, write: if true;
    }
  }
}
